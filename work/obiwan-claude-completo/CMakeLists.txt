cmake_minimum_required(VERSION 3.20)
project(obiwan_recomp C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$" OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "This bring-up launcher targets x86-64")
endif()
if(WIN32 AND NOT MSVC)
  message(FATAL_ERROR "The Windows launcher is built with MSVC")
elseif(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "This bring-up launcher targets x86-64 Linux or Windows")
endif()
set(_runtime_default "${CMAKE_CURRENT_SOURCE_DIR}/../xbox-linux/third_party/xboxrecomp")
if(DEFINED ENV{XBOXRECOMP_DIR} AND NOT "$ENV{XBOXRECOMP_DIR}" STREQUAL "")
  set(_runtime_default "$ENV{XBOXRECOMP_DIR}")
endif()
set(XBOXRECOMP_DIR "${_runtime_default}" CACHE PATH "Path to separately cloned Xbox Linux runtime dependency")
if(NOT WIN32)
  find_package(OpenSSL REQUIRED)
endif()
add_subdirectory("${XBOXRECOMP_DIR}" "${CMAKE_BINARY_DIR}/runtime")
file(GLOB GENERATED CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen/*.c")
if(NOT GENERATED)
  message(FATAL_ERROR "Run tools/generate_obiwan.py before configuring this project")
endif()
add_executable(obiwan_recomp src/main.c src/manual.c src/perfil.c ${GENERATED})
target_include_directories(obiwan_recomp PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/src/recomp/gen")
if(WIN32)
  # CNG hashes the XBE; dbghelp names the host function in a fatal report.
  target_link_libraries(obiwan_recomp PRIVATE xboxrecomp bcrypt dbghelp)
  target_compile_definitions(obiwan_recomp PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
  # MSVC does not optimise on strict aliasing or signed overflow; the
  # generated files need the larger section limit.
  target_compile_options(obiwan_recomp PRIVATE /bigobj)
  target_link_options(obiwan_recomp PRIVATE /LARGEADDRESSAWARE)
else()
  target_link_libraries(obiwan_recomp PRIVATE xboxrecomp OpenSSL::Crypto)
  # Start with debug-friendly code; guest semantics must not depend on host aliasing rules.
  target_compile_options(obiwan_recomp PRIVATE -fno-strict-aliasing -fwrapv)
  target_link_options(obiwan_recomp PRIVATE -rdynamic)
endif()
